<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class EnsureAdminRole
{
    /**
     * Handle an incoming request.
     *
     * @param  \Closure(\Illuminate\Http\Request): (\Symfony\Component\HttpFoundation\Response)  $next
     */
    public function handle(Request $request, Closure $next): Response
    {
        if (!auth()->check()) {
            return redirect('/login');
        }

        if (!auth()->user()->isAdmin()) {
            return redirect('/login')->withErrors(['email' => 'Access denied. Admin privileges required.']);
        }

        return $next($request);
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\BrakeEvent;
use App\Models\AccidentProneArea;
use App\Support\BarangayLocator;
use App\Services\EmergencyBrakeApaTracker;
use Carbon\Carbon;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;
use Illuminate\Support\Facades\Validator;

class DashboardController extends Controller
{
    public function dashboardOverview()
    {
        $municipalities = ['Bantayan', 'Madridejos', 'Santa Fe'];
        $classifications = [
            'Soft' => 'soft',
            'Moderate' => 'medium',
            'Hard' => 'hard',
            'Emergency' => 'emergency',
        ];
        $severityByMunicipality = [];
        $municipalityTotals = array_fill_keys($municipalities, 0);

        foreach ($classifications as $classification) {
            $severityByMunicipality[$classification] = array_fill_keys($municipalities, 0);
        }

        $datasetInfo = null;
        if (Storage::disk('local')->exists('settings/dashboard-dataset.json')) {
            $savedDatasetInfo = json_decode(Storage::disk('local')->get('settings/dashboard-dataset.json'), true);
            if (is_array($savedDatasetInfo) && !empty($savedDatasetInfo['file_name'])) {
                $datasetInfo = $savedDatasetInfo;
            }
        }

        $mappedEvents = $datasetInfo
            ? BrakeEvent::query()
                ->whereNotNull('latitude')
                ->whereNotNull('longitude')
                ->whereBetween('latitude', [-90, 90])
                ->whereBetween('longitude', [-180, 180])
                ->whereIn('municipality', $municipalities)
                ->get(['municipality', 'brake_classification'])
            : collect();

        foreach ($mappedEvents as $event) {
            $municipality = collect($municipalities)->first(
                fn ($name) => strtolower($name) === strtolower(trim((string) $event->municipality))
            );
            $classification = strtolower(trim((string) $event->brake_classification));

            if ($municipality === null || !isset($severityByMunicipality[$classification])) {
                continue;
            }

            $severityByMunicipality[$classification][$municipality]++;
            $municipalityTotals[$municipality]++;
        }

        $mappedEventCount = array_sum($municipalityTotals);
        $datasetEventCount = $datasetInfo ? BrakeEvent::count() : 0;
        $mapTilerKey = config('services.maptiler.key');
        return view('dashboard.index', [
            'municipalities' => $municipalities,
            'classifications' => array_keys($classifications),
            'severityByMunicipality' => $severityByMunicipality,
            'municipalityTotals' => $municipalityTotals,
            'mappedEventCount' => $mappedEventCount,
            'datasetEventCount' => $datasetEventCount,
            'datasetInfo' => $datasetInfo,
            'mapTilerKey' => $mapTilerKey,
            'proximityTargets' => $this->proximityTargets(),
        ]);
    }

    public function nearbyAlertTargets(): JsonResponse
    {
        return response()->json(['targets' => $this->proximityTargets()]);
    }

    private function proximityTargets()
    {
        $municipalities = ['Bantayan', 'Madridejos', 'Santa Fe'];
        $proximityBrakeEvents = BrakeEvent::query()
            ->whereNotNull('latitude')
            ->whereNotNull('longitude')

    private function parseBrakeEventTimestamp(string $timestamp): ?Carbon
    {
        if (is_numeric($timestamp) && (float) $timestamp > 10000) {
            return Carbon::createFromTimestampUTC((int) round(((float) $timestamp - 25569) * 86400));
        }

        foreach (['!d/m/Y H:i:s', '!d/m/Y H:i'] as $format) {
            try {
                $parsedTimestamp = Carbon::createFromFormat($format, $timestamp);
                if ($parsedTimestamp !== false) {
                    return $parsedTimestamp;
                }
            } catch (\Exception $exception) {
                continue;
            }
        }

        try {
            return Carbon::parse($timestamp);
        } catch (\Exception $exception) {
            return null;
        }
    }

    private function readXlsxRows(string $path): array
    {
        $archive = new \ZipArchive();
        if ($archive->open($path) !== true) {
            throw new \RuntimeException('Unable to open the uploaded Excel workbook.');
        }

        try {
            $sharedStrings = [];
            $parseXml = function (string $xml, string $errorMessage): \SimpleXMLElement {
                $previousErrorMode = libxml_use_internal_errors(true);
                try {
                    $document = simplexml_load_string($xml);
                    if ($document === false) {
                        throw new \RuntimeException($errorMessage);
                    }

                    return $document;
                } finally {
                    libxml_clear_errors();
                    libxml_use_internal_errors($previousErrorMode);
                }
            };

            $sharedStringsXml = $archive->getFromName('xl/sharedStrings.xml');
            if ($sharedStringsXml !== false) {
                $sharedStringsDocument = $parseXml($sharedStringsXml, 'The Excel workbook contains invalid shared strings.');
                $sharedStringsDocument->registerXPathNamespace('x', 'http://schemas.openxmlformats.org/spreadsheetml/2006/main');
                foreach ($sharedStringsDocument->xpath('//x:si') as $string) {
                    $string->registerXPathNamespace('x', 'http://schemas.openxmlformats.org/spreadsheetml/2006/main');
                    $sharedStrings[] = implode('', array_map('strval', $string->xpath('.//x:t')));
                }
            }

            $worksheetXml = $archive->getFromName('xl/worksheets/sheet1.xml');
            if ($worksheetXml === false) {
                throw new \RuntimeException('The Excel workbook does not contain a readable first worksheet.');
            }
            $worksheet = $parseXml($worksheetXml, 'The Excel workbook contains invalid worksheet data.');
            $worksheet->registerXPathNamespace('x', 'http://schemas.openxmlformats.org/spreadsheetml/2006/main');

            $rows = [];
            foreach ($worksheet->xpath('//x:sheetData/x:row') as $row) {
                $row->registerXPathNamespace('x', 'http://schemas.openxmlformats.org/spreadsheetml/2006/main');
                $values = [];
                foreach ($row->xpath('./x:c') as $cell) {
                    $reference = (string) $cell['r'];
                    preg_match('/^[A-Z]+/', $reference, $columnMatch);
                    if (!$columnMatch) {
                        continue;
                    }

                    $columnIndex = 0;
                    foreach (str_split($columnMatch[0]) as $letter) {
                        $columnIndex = ($columnIndex * 26) + (ord($letter) - ord('A') + 1);
                    }
                    $columnIndex--;

                    $type = (string) $cell['t'];
                    $value = (string) ($cell->v ?? '');
                    if ($type === 's') {
                        $value = $sharedStrings[(int) $value] ?? '';
                    } elseif ($type === 'inlineStr') {
                        $cell->registerXPathNamespace('x', 'http://schemas.openxmlformats.org/spreadsheetml/2006/main');
                        $value = implode('', array_map('strval', $cell->xpath('.//x:t')));
                    }
                    $values[$columnIndex] = $value;
                }

                if ($values) {
                    ksort($values);
                    $rows[] = array_replace(array_fill(0, max(array_keys($values)) + 1, ''), $values);
                }
            }

            return $rows;
        } finally {
            $archive->close();
        }
    }

namespace App\Services;

use App\Models\AccidentProneArea;
use App\Models\BrakeEvent;
use App\Models\EmergencyBrakeApaCluster;
use App\Support\BarangayLocator;
use Illuminate\Support\Facades\DB;

class EmergencyBrakeApaTracker
{
    private const APA_CREATION_THRESHOLD = 5;
    private const CLUSTER_RADIUS_METERS = 100;
    private const DUPLICATE_CHECK_RADIUS_METERS = 5000;
    private const ALLOWED_MUNICIPALITIES = ['Bantayan', 'Madridejos', 'Santa Fe'];

    public function __construct(private readonly BarangayLocator $barangayLocator)
    {
    }

    public function process(BrakeEvent $event): void
    {
        if (strtolower(trim((string) $event->brake_classification)) !== 'emergency') {
            return;
        }

        DB::transaction(function () use ($event): void {
            $lockedEvent = BrakeEvent::query()->lockForUpdate()->find($event->id);
            if (!$lockedEvent || $lockedEvent->apa_processed_at !== null) {
                return;
            }

            $latitude = (float) $lockedEvent->latitude;
            $longitude = (float) $lockedEvent->longitude;
            $location = $this->barangayLocator->locate($latitude, $longitude);
            if (
                $latitude < -90 || $latitude > 90
                || $longitude < -180 || $longitude > 180
                || $location === null
                || !in_array($location['municipality'], self::ALLOWED_MUNICIPALITIES, true)
            ) {
                $this->markProcessed($lockedEvent);
                return;
            }

            $areas = AccidentProneArea::query()->lockForUpdate()->get();
            $nearestArea = $areas
                ->map(fn (AccidentProneArea $area) => [
                    'area' => $area,
                    'distance' => $this->distanceInMeters(
                        $latitude,
                        $longitude,
                        (float) $area->latitude,
                        (float) $area->longitude
                    ),
                ])
                ->sortBy('distance')
                ->first();

            if ($nearestArea && $nearestArea['distance'] <= self::CLUSTER_RADIUS_METERS) {
                $area = $nearestArea['area'];
                $area->increment('emergency_hit_count');
                $this->markProcessed($lockedEvent);
                return;
            }

            if ($nearestArea && $nearestArea['distance'] <= self::DUPLICATE_CHECK_RADIUS_METERS) {
                $this->markProcessed($lockedEvent);
                return;
            }

            $cluster = EmergencyBrakeApaCluster::query()
                ->whereNull('accident_prone_area_id')
                ->lockForUpdate()
                ->get()
                ->map(fn (EmergencyBrakeApaCluster $candidate) => [
                    'cluster' => $candidate,
                    'distance' => $this->distanceInMeters(
                        $latitude,
                        $longitude,
                        (float) $candidate->latitude,
                        (float) $candidate->longitude
                    ),
                ])
                ->filter(fn (array $candidate) => $candidate['distance'] <= self::CLUSTER_RADIUS_METERS)
                ->sortBy('distance')
                ->first();

            if ($cluster) {
                $candidate = $cluster['cluster'];
                $candidate->increment('hit_count');
                $candidate->refresh();
            } else {
                $candidate = EmergencyBrakeApaCluster::create([
                    'latitude' => $latitude,
                    'longitude' => $longitude,
                    'hit_count' => 1,
                ]);
            }

            if ($candidate->hit_count >= self::APA_CREATION_THRESHOLD) {
                $clusterLocation = $this->barangayLocator->locate(
                    (float) $candidate->latitude,
                    (float) $candidate->longitude
                );
                if ($clusterLocation === null) {
                    $this->markProcessed($lockedEvent);
                    return;
                }
                $area = AccidentProneArea::create([
                    'latitude' => $candidate->latitude,
                    'longitude' => $candidate->longitude,
                    'barangay' => $clusterLocation['barangay'],
                    'municipality' => $clusterLocation['municipality'],
                    'province' => 'Cebu',
                    'country' => 'Philippines',
                    'place' => "{$clusterLocation['barangay']}, {$clusterLocation['municipality']}, Cebu, Philippines",
                    'reason' => 'Auto-created after 5 emergency brake events within 100 m.',
                    'emergency_hit_count' => $candidate->hit_count,
                    'is_auto_generated' => true,
                ]);
                $candidate->update(['accident_prone_area_id' => $area->id]);
            }

            $this->markProcessed($lockedEvent);
        });
    }

    public function processMany(array $eventIds): void
    {
        BrakeEvent::query()
            ->whereIn('id', array_values(array_unique($eventIds)))
            ->orderBy('timestamp')
            ->orderBy('id')
            ->get()
            ->each(fn (BrakeEvent $event) => $this->process($event));
    }

    private function markProcessed(BrakeEvent $event): void
    {
        $event->apa_processed_at = now();
        $event->save();
    }

    private function distanceInMeters(float $latitudeA, float $longitudeA, float $latitudeB, float $longitudeB): float
    {
        $radians = static fn (float $degrees): float => $degrees * M_PI / 180;
        $latitudeDelta = $radians($latitudeB - $latitudeA);
        $longitudeDelta = $radians($longitudeB - $longitudeA);
        $haversine = sin($latitudeDelta / 2) ** 2
            + cos($radians($latitudeA)) * cos($radians($latitudeB))
            * sin($longitudeDelta / 2) ** 2;

        return 6371000 * 2 * atan2(sqrt($haversine), sqrt(1 - $haversine));
    }

use RuntimeException;

class BarangayLocator
{
    private const MUNICIPALITIES = [
        'bgy-702209000.json' => 'Bantayan',
        'bgy-702228000.json' => 'Madridejos',
        'bgy-702244000.json' => 'Santa Fe',

    private static ?array $barangays = null;

    public function locate(float $latitude, float $longitude): ?array
    {
        foreach ($this->barangays() as $municipality => $barangays) {
            foreach ($barangays as $barangay) {
                [$minLongitude, $minLatitude, $maxLongitude, $maxLatitude] = $barangay['bounds'];
                if (
                    $longitude < $minLongitude
                    || $longitude > $maxLongitude
                    || $latitude < $minLatitude
                    || $latitude > $maxLatitude
                ) {
                    continue;
                }

                if ($this->containsPoint($longitude, $latitude, $barangay['geometry'])) {
                    return [
                        'municipality' => $municipality,
                        'barangay' => $this->displayName($municipality, $barangay['name']),
                    ];
                }
            }
        }

        return null;
    }

    public function fillLocation(float $latitude, float $longitude, ?string $municipality = null, ?string $barangay = null): array
    {
        $location = $this->locate($latitude, $longitude);
        if ($location !== null) {
            return $location;
        }

        $municipality = trim((string) ($municipality ?? ''));
        if ($municipality === '' || !in_array($municipality, ['Bantayan', 'Madridejos', 'Santa Fe'], true)) {
            $municipality = 'Madridejos';
        }

        $barangay = trim((string) ($barangay ?? ''));

        return [
            'municipality' => $municipality,
            'barangay' => $barangay === '' ? 'Unknown' : $barangay,
        ];
    }

    public function all(): array
    {
        $locations = [];
        foreach ($this->barangays() as $municipality => $barangays) {
            $locations[$municipality] = array_values(array_map(
                fn (array $barangay) => $this->displayName($municipality, $barangay['name']),
                $barangays
            ));
            sort($locations[$municipality], SORT_NATURAL | SORT_FLAG_CASE);
        }

        return $locations;
    }

    private function displayName(string $municipality, string $name): string
    {
        if ($municipality === 'Bantayan' && in_array($name, ['Bantigue', 'Binaobao', 'Suba', 'Ticad'], true)) {
            return "{$name} (Poblacion)";
        }

        return $name;
    }

    private function barangays(): array
    {
        if (self::$barangays !== null) {
            return self::$barangays;
        }

        $barangays = [];
        foreach (self::MUNICIPALITIES as $fileName => $municipality) {
            $path = resource_path('geojson/' . $fileName);
            $contents = file_get_contents($path);
            if ($contents === false) {
                throw new RuntimeException("Unable to read barangay boundaries: {$path}");
            }

            $collection = json_decode($contents, true, 512, JSON_THROW_ON_ERROR);
            foreach ($collection['features'] ?? [] as $feature) {
                $geometry = $feature['geometry'] ?? null;
                $name = trim((string) ($feature['properties']['adm4_en'] ?? ''));
                if ($name === '' || !is_array($geometry)) {
                    continue;
                }

                $bounds = $this->bounds($geometry['coordinates'] ?? []);
                if ($bounds !== null) {
                    $barangays[$municipality][] = [
                        'name' => $name,
                        'geometry' => $geometry,
                        'bounds' => $bounds,
                    ];
                }
            }
        }

        return self::$barangays = $barangays;
    }

    private function bounds(array $coordinates): ?array
    {
        $longitudes = [];
        $latitudes = [];
        $collect = function (array $values) use (&$collect, &$longitudes, &$latitudes): void {
            if (
                count($values) >= 2
                && is_numeric($values[0])
                && is_numeric($values[1])
            ) {
                $longitudes[] = (float) $values[0];
                $latitudes[] = (float) $values[1];
                return;
            }

            foreach ($values as $value) {
                if (is_array($value)) {
                    $collect($value);
                }
            }
        };
        $collect($coordinates);

        if ($longitudes === []) {
            return null;
        }

        return [
            min($longitudes),
            min($latitudes),
            max($longitudes),
            max($latitudes),
        ];
    }

    private function containsPoint(float $longitude, float $latitude, array $geometry): bool
    {
        $polygons = match ($geometry['type'] ?? null) {
            'Polygon' => [$geometry['coordinates'] ?? []],
            'MultiPolygon' => $geometry['coordinates'] ?? [],
            default => [],
        };

        foreach ($polygons as $rings) {
            if (!isset($rings[0]) || !$this->insideRing($longitude, $latitude, $rings[0])) {
                continue;
            }

            $insideHole = false;
            foreach (array_slice($rings, 1) as $hole) {
                if ($this->insideRing($longitude, $latitude, $hole)) {
                    $insideHole = true;
                    break;
                }
            }

            if (!$insideHole) {
                return true;
            }
        }

        return false;
    }

    private function insideRing(float $longitude, float $latitude, array $ring): bool
    {
        $inside = false;
        $count = count($ring);
        for ($index = 0, $previous = $count - 1; $index < $count; $previous = $index++) {
            $x1 = (float) $ring[$index][0];
            $y1 = (float) $ring[$index][1];
            $x2 = (float) $ring[$previous][0];
            $y2 = (float) $ring[$previous][1];

            $cross = ($longitude - $x1) * ($y2 - $y1) - ($latitude - $y1) * ($x2 - $x1);
            if (
                abs($cross) < 1.0e-10
                && $longitude >= min($x1, $x2)
                && $longitude <= max($x1, $x2)
                && $latitude >= min($y1, $y2)
                && $latitude <= max($y1, $y2)
            ) {
                return true;
            }

            $crossesLatitude = (($y1 > $latitude) !== ($y2 > $latitude));
            if ($crossesLatitude && $longitude < ($x2 - $x1) * ($latitude - $y1) / ($y2 - $y1) + $x1) {
                $inside = !$inside;
            }
        }

        return $inside;
    }
}

@push('scripts')
<script>
    document.addEventListener('DOMContentLoaded', () => {
        const municipalities = @json($municipalities);
        const municipalityColors = @json($municipalityColors);
        const mapTilerKey = @json($mapTilerKey);
        let proximityTargets = @json($proximityTargets);
        const defaultCenter = [11.2200, 123.7600];
        const mapElement = document.getElementById('dashboard-map');
        const mapDebug = document.getElementById('dashboard-map-debug');
        const emptyState = document.getElementById('map-empty-state');
        const gpsButton = document.getElementById('use-gps');
        const gpsStatus = document.getElementById('location-status');
        const latitudeOutput = document.getElementById('device-latitude');
        const longitudeOutput = document.getElementById('device-longitude');
        const accuracyOutput = document.getElementById('device-accuracy');
        const directionOutput = document.getElementById('device-direction');
        const compassStatus = document.getElementById('compass-status');
        const compassIndicator = document.getElementById('compass-indicator');
        const compassNeedle = document.getElementById('compass-needle');
        let gpsMarker = null;
        let gpsWatchId = null;
        let currentHeading = 0;
        let notificationsEnabled = false;
        const nearbyTargetIds = new Set();
        const activeNearbyNotifications = new Set();
        const gpsSessionKey = 'brakesafe-dashboard-gps-enabled';
        const notificationsSessionKey = 'brakesafe-dashboard-notifications-enabled';
        const readSessionFlag = (key) => {
            try {
                return sessionStorage.getItem(key) === 'true';
            } catch (error) {
                console.error('Unable to restore a dashboard preference.', error);
                return false;
            }
        };
        const writeSessionFlag = (key, enabled) => {
            try {
                if (enabled) {
                    sessionStorage.setItem(key, 'true');
                } else {
                    sessionStorage.removeItem(key);
                }
            } catch (error) {
                console.error('Unable to save a dashboard preference.', error);
            }
        };
        const resumeGpsAfterReload = readSessionFlag(gpsSessionKey);
        const resumeNotificationsAfterReload = readSessionFlag(notificationsSessionKey);

        const updateDeviceClock = () => {
            const now = new Date();
            document.getElementById('device-local-time').textContent = now.toLocaleTimeString([], { hour: 'numeric', minute: '2-digit' });
            document.getElementById('device-local-date').textContent = now.toLocaleDateString([], { year: 'numeric', month: 'long', day: 'numeric' });
        };
        updateDeviceClock();
        window.setInterval(updateDeviceClock, 1000);

        const map = typeof L !== 'undefined' && mapElement
            ? L.map(mapElement, { zoomControl: false }).setView(defaultCenter, 11)
            : null;
        let mapTiles = null;
        const isLightTheme = () => document.documentElement.dataset.adminTheme === 'light';
        const gpsArrowIcon = () => {
            const color = isLightTheme() ? '#2563eb' : '#dc2626';
            return L.divIcon({
                className: 'gps-location-icon',
                html: `<div class="gps-location-arrow" style="transform:rotate(${currentHeading}deg)"><svg viewBox="0 0 24 24" aria-hidden="true"><path d="M12 1.5 21 22l-9-4.2L3 22 12 1.5Z" fill="${color}" stroke="white" stroke-width="1.7" stroke-linejoin="round"/></svg></div>`,
                iconSize: [40, 40],
                iconAnchor: [20, 20],
            });
        };
        const updateGpsMarkerIcon = () => {
            if (gpsMarker) gpsMarker.setIcon(gpsArrowIcon());
        };

        if (map) {
            window.BrakeSafeMapAttribution?.move(map, 'dashboard-map-attribution');
            window.BrakeSafeMapLayout?.observe(map, mapElement);
            if (mapTilerKey) {
                mapTiles = L.tileLayer(
                    `https://api.maptiler.com/maps/${isLightTheme() ? 'streets-v2' : 'streets-v2-dark'}/{z}/{x}/{y}.png?key=${encodeURIComponent(mapTilerKey)}`,
                    {
                        maxZoom: 19,
                        attribution: '&copy; <a href="https://www.maptiler.com/copyright/">MapTiler</a> &copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors',
                    }
                );
                mapTiles.once('tileload', () => { mapDebug.style.display = 'none'; });
                mapTiles.once('tileerror', () => {
                    mapDebug.textContent = 'MapTiler could not load map tiles. Check that MAPTILER_API_KEY is valid and allows this site.';
                    mapDebug.style.display = 'flex';
                    console.error('MapTiler tile request failed. Check the API key and its allowed domains.');
                });
                mapTiles.addTo(map);
            }

            window.BrakeSafeMapLayout?.restoreView(map, 'brakesafe-map-dashboard');
            window.setTimeout(() => map.invalidateSize(), 100);
        } else if (mapElement) {
            mapElement.setAttribute('aria-label', 'Map unavailable because the map library did not load');
        }

        document.getElementById('map-zoom-in').addEventListener('click', () => map?.zoomIn());
        document.getElementById('map-zoom-out').addEventListener('click', () => map?.zoomOut());
        window.addEventListener('brakesafe:theme-change', (event) => {
            if (mapTiles && mapTilerKey) {
                const style = event.detail?.isLight ? 'streets-v2' : 'streets-v2-dark';
                mapTiles.setUrl(`https://api.maptiler.com/maps/${style}/{z}/{x}/{y}.png?key=${encodeURIComponent(mapTilerKey)}`);
            }
            updateGpsMarkerIcon();
        });

        const compassDirections = ['N', 'NE', 'E', 'SE', 'S', 'SW', 'W', 'NW'];
        let compassRequested = false;
        const startCompass = async () => {
            if (compassRequested) return;
            compassRequested = true;
            if (typeof DeviceOrientationEvent === 'undefined') {
                compassStatus.textContent = 'Compass unavailable on this device';
                return;
            }

            if (typeof DeviceOrientationEvent.requestPermission === 'function') {
                try {
                    const permission = await DeviceOrientationEvent.requestPermission();
                    if (permission !== 'granted') {
                        compassStatus.textContent = 'Compass permission not granted';
                        return;
                    }
                } catch (error) {
                    compassStatus.textContent = 'Compass permission unavailable';
                    console.error('Unable to request compass permission', error);
                    return;
                }
            }

            compassStatus.textContent = 'Waiting for compass data';
            window.addEventListener('deviceorientationabsolute', updateCompass);
            window.addEventListener('deviceorientation', updateCompass);
        };

        function updateCompass(event) {
            const heading = Number.isFinite(event.webkitCompassHeading)
                ? event.webkitCompassHeading
                : Number.isFinite(event.alpha) ? (360 - event.alpha) % 360 : null;
            if (heading === null) return;
            currentHeading = heading;

            const direction = compassDirections[Math.round(heading / 45) % compassDirections.length];
            directionOutput.textContent = `${direction} (${Math.round(heading)}°)`;
            directionOutput.classList.remove('text-slate-400');
            directionOutput.classList.add('text-slate-200');
            compassStatus.textContent = 'Compass active';
            compassIndicator.setAttribute('aria-label', `Compass heading ${direction}, ${Math.round(heading)} degrees`);
            compassIndicator.classList.remove('text-slate-700');
            compassIndicator.classList.add('text-cyan-300');
            compassNeedle.classList.remove('opacity-35');
            compassNeedle.style.transform = `rotate(${heading}deg)`;
            updateGpsMarkerIcon();
        }

        const distanceInMeters = (latitudeA, longitudeA, latitudeB, longitudeB) => {
            const radians = (degrees) => degrees * Math.PI / 180;
            const latitudeDelta = radians(latitudeB - latitudeA);
            const longitudeDelta = radians(longitudeB - longitudeA);
            const haversine = Math.sin(latitudeDelta / 2) ** 2
                + Math.cos(radians(latitudeA)) * Math.cos(radians(latitudeB))
                * Math.sin(longitudeDelta / 2) ** 2;
            return 6371000 * 2 * Math.atan2(Math.sqrt(haversine), Math.sqrt(1 - haversine));
        };

        const nearbyAlertQueue = [];
        let processingNearbyAlerts = false;
        const processNearbyAlertQueue = async () => {
            if (processingNearbyAlerts) return;
            processingNearbyAlerts = true;
            try {
                while (nearbyAlertQueue.length > 0 && notificationsEnabled && Notification.permission === 'granted') {
                    const target = nearbyAlertQueue.shift();
                    const alert = window.BrakeSafeNearbyAlerts?.format(target, target.distance) ?? {
                        title: `Nearby ${target.type} Brake Event in ${target.place || 'saved location'}`,
                        body: `is about ${Math.round(target.distance)} m away.`,
                    };
                    let notification = null;

                    try {
                        notification = new Notification(alert.title, {
                            body: alert.body,
                            tag: `brakesafe-nearby-${target.id}`,
                        });
                        activeNearbyNotifications.add(notification);
                        notification.onclose = () => activeNearbyNotifications.delete(notification);
                        notification.onclick = () => {
                            window.focus();
                            map?.setView([target.latitude, target.longitude], 16);
                            notification.close();
                        };
                    } catch (error) {
                        console.error('Unable to display a nearby browser notification.', error);
                    }

                    if (window.Swal) {
                        await window.Swal.fire({
                            toast: true,
                            position: 'top-end',
                            icon: 'warning',
                            title: alert.title,
                            text: alert.body,
                            showConfirmButton: false,
                            timer: 3500,
                            timerProgressBar: true,
                        });
                    } else {
                        await new Promise((resolve) => window.setTimeout(resolve, 3500));
                    }

                    notification?.close();
                }
            } finally {
                processingNearbyAlerts = false;
                if (nearbyAlertQueue.length > 0 && notificationsEnabled && Notification.permission === 'granted') {
                    void processNearbyAlertQueue();
                }
            }
        };

        const notifyNearbyTargets = ({ latitude, longitude, accuracy }) => {
            if (!notificationsEnabled || Notification.permission !== 'granted') return;
            if (accuracy > 400) {
                gpsStatus.textContent = `GPS accuracy is too low (${Math.round(accuracy)} m) for reliable 400 m alerts.`;
                return;
            }

            const newlyNearbyTargets = [];
            proximityTargets.forEach((target) => {
                const distance = distanceInMeters(latitude, longitude, target.latitude, target.longitude);
                const targetId = String(target.id);
                if (distance <= 400 && !nearbyTargetIds.has(targetId)) {
                    nearbyTargetIds.add(targetId);
                    newlyNearbyTargets.push({ ...target, distance });
                } else if (distance > 550) {
                    nearbyTargetIds.delete(targetId);
                }
            });

            if (newlyNearbyTargets.length === 0) return;

            newlyNearbyTargets.sort((first, second) => first.distance - second.distance);
            nearbyAlertQueue.push(...newlyNearbyTargets);
            void processNearbyAlertQueue();
        };

        window.addEventListener('brakesafe:live-update', (event) => {
            if (event.detail?.pageId !== 'dashboard-root' || !Array.isArray(event.detail.data?.proximityTargets)) return;
            proximityTargets = event.detail.data.proximityTargets;
            const currentTargetIds = new Set(proximityTargets.map((target) => String(target.id)));
            nearbyTargetIds.forEach((targetId) => {
                if (!currentTargetIds.has(targetId)) nearbyTargetIds.delete(targetId);
            });
        });

        const handleGpsPosition = (position, centerMap = false) => {
            const { latitude, longitude, accuracy, heading } = position.coords;
            if (Number.isFinite(heading)) currentHeading = heading;
            latitudeOutput.textContent = latitude.toFixed(6);
            longitudeOutput.textContent = longitude.toFixed(6);
            accuracyOutput.textContent = `${Math.round(accuracy)} m`;
            gpsStatus.textContent = 'GPS tracking active.';

            if (map) {
                if (!gpsMarker) {
                    gpsMarker = L.marker([latitude, longitude], {
                        icon: gpsArrowIcon(),
                        keyboard: true,
                        title: 'Your current location',
                        zIndexOffset: 1000,
                    }).addTo(map).bindPopup('Your current location');
                    emptyState.classList.add('hidden');
                } else {
                    gpsMarker.setLatLng([latitude, longitude]);
                    updateGpsMarkerIcon();
                }
                if (centerMap) map.setView([latitude, longitude], 14);
            }

            notifyNearbyTargets({ latitude, longitude, accuracy });
        };

        const handleGpsError = (error) => {
            const messages = {
                1: 'Location permission was denied.',
                2: 'Device location is unavailable.',
                3: 'Location request timed out.',
            };
            gpsStatus.textContent = messages[error.code] || 'Unable to get device location.';
            if (error.code === 1 && gpsWatchId !== null) {
                navigator.geolocation.clearWatch(gpsWatchId);
                gpsWatchId = null;
                writeSessionFlag(gpsSessionKey, false);
                gpsButton.setAttribute('aria-pressed', 'false');
                gpsButton.setAttribute('aria-label', 'Turn on GPS');
                gpsButton.querySelector('span').textContent = 'Use device GPS';
                if (notificationsEnabled) {
                    setNearbyNotificationsEnabled(false);
                    mapNote.textContent = 'Location permission was denied; nearby notifications are off.';
                }
            }
            if (gpsWatchId === null) gpsButton.querySelector('span').textContent = 'Use device GPS';
            console.error('Unable to track device location.', error);
        };

        const startGpsTracking = () => {
            if (!navigator.geolocation) {
                gpsStatus.textContent = 'Device location is not available in this browser.';
                return false;
            }
            if (gpsWatchId !== null) return true;

            gpsButton.disabled = true;
            gpsButton.querySelector('span').textContent = 'Requesting GPS…';
            gpsStatus.textContent = 'Waiting for location permission.';
            startCompass();
            gpsWatchId = navigator.geolocation.watchPosition((position) => {
                const firstFix = !gpsMarker;
                handleGpsPosition(position, firstFix);
                gpsButton.disabled = false;
            }, (error) => {
                handleGpsError(error);
                gpsButton.disabled = false;
            }, { enableHighAccuracy: true, timeout: 20000, maximumAge: 5000 });
            gpsButton.setAttribute('aria-pressed', 'true');
            gpsButton.setAttribute('aria-label', 'Turn off GPS');
            gpsButton.querySelector('span').textContent = 'Turn off GPS';
            writeSessionFlag(gpsSessionKey, true);
            return true;
        };

        const confirmAndStartGpsTracking = async () => {
            if (gpsWatchId !== null) {
                return true;
            }
            if (!navigator.geolocation) {
                gpsStatus.textContent = 'Device location is not available in this browser.';
                return false;
            }

            if (window.Swal) {
                const confirmation = await window.Swal.fire({
                    icon: 'question',
                    title: 'Turn on GPS tracking?',
                    showCancelButton: true,
                    confirmButtonText: 'Continue',
                    cancelButtonText: 'Not now',
                    reverseButtons: true,
                });
                if (!confirmation.isConfirmed) {
                    return false;
                }
            }

            return startGpsTracking();
        };

        const nearbyButton = document.getElementById('nearby-notifications-toggle');
        const mapNote = document.getElementById('dashboard-map-note');
        const setNearbyNotificationsEnabled = (enabled, title = null) => {
            notificationsEnabled = enabled;
            writeSessionFlag(notificationsSessionKey, enabled);
            if (!enabled) {
                nearbyAlertQueue.length = 0;
                activeNearbyNotifications.forEach((notification) => notification.close());
                activeNearbyNotifications.clear();
            }
            nearbyButton.setAttribute('aria-pressed', String(enabled));
            nearbyButton.setAttribute('aria-label', enabled ? 'Disable nearby notifications' : 'Enable nearby notifications');
            nearbyButton.title = title ?? (enabled ? 'Nearby notifications are on' : 'Nearby notifications are off');
            nearbyButton.querySelector('span').textContent = enabled ? 'Nearby notifications on' : 'Nearby notifications off';
            nearbyButton.classList.toggle('nearby-notifications-enabled', enabled);
            nearbyButton.querySelector('svg').innerHTML = enabled
                ? '<path d="M18 8a6 6 0 0 0-12 0c0 7-3 7-3 9h18c0-2-3-2-3-9"/><path d="M10 21h4"/><path d="m9 11 2 2 4-4"/>'
                : '<path d="M18 8a6 6 0 0 0-12 0c0 7-3 7-3 9h18c0-2-3-2-3-9"/><path d="M10 21h4"/><path d="m3 3 18 18"/>';
        };
        const stopGpsTracking = () => {
            if (gpsWatchId !== null) {
                navigator.geolocation.clearWatch(gpsWatchId);
                gpsWatchId = null;
            }
            writeSessionFlag(gpsSessionKey, false);
            if (gpsMarker && map) {
                map.removeLayer(gpsMarker);
                gpsMarker = null;
            }
            emptyState.classList.remove('hidden');
            latitudeOutput.textContent = '--';
            longitudeOutput.textContent = '--';
            accuracyOutput.textContent = '--';
            gpsButton.querySelector('span').textContent = 'Use device GPS';
            gpsButton.setAttribute('aria-pressed', 'false');
            gpsButton.setAttribute('aria-label', 'Turn on GPS');
            gpsStatus.textContent = 'GPS tracking stopped.';
            if (notificationsEnabled) {
                setNearbyNotificationsEnabled(false);
                mapNote.textContent = 'GPS tracking stopped; nearby notifications are off.';
            }
            nearbyTargetIds.clear();
        };

        gpsButton.setAttribute('aria-label', 'Turn on GPS');
        gpsButton.setAttribute('aria-pressed', 'false');
        gpsButton.addEventListener('click', async () => {
            if (gpsWatchId !== null) {
                if (window.Swal) {
                    const confirmation = await window.Swal.fire({
                        icon: 'warning',
                        title: 'Turn off GPS tracking?',
                        showCancelButton: true,
                        confirmButtonText: 'Turn off GPS',
                        cancelButtonText: 'Keep tracking',
                        reverseButtons: true,
                    });
                    if (!confirmation.isConfirmed) {
                        return;
                    }
                }
                stopGpsTracking();
            } else {
                await confirmAndStartGpsTracking();
            }
        });

        nearbyButton.addEventListener('click', async () => {
            if (notificationsEnabled) {
                setNearbyNotificationsEnabled(false);
                mapNote.textContent = 'Nearby notifications are off. Browser permission remains enabled; change it in site settings to revoke permission.';
                if (window.Swal) {
                    await window.Swal.fire({
                        icon: 'info',
                        title: 'Nearby notifications turned off',
                        text: 'BrakeSafe will stop sending alerts and close any visible alerts. Browser notification permission cannot be revoked by a webpage; change it in your browser site settings if you also want to revoke permission.',
                    });
                }
                return;
            }

            if (!('Notification' in window)) {
                gpsStatus.textContent = 'Browser notifications are not available.';
                if (window.Swal) {
                    await window.Swal.fire({
                        icon: 'error',
                        title: 'Notifications unavailable',
                        text: 'This browser does not support desktop notifications.',
                    });
                }
                return;
            }

            if (window.Swal) {
                const confirmation = await window.Swal.fire({
                    icon: 'question',
                    title: 'Enable nearby notifications?',
                    text: 'Allow browser notifications for hard and emergency brake events and saved accident-prone areas within 400 meters. GPS tracking and this dashboard must remain active.',
                    showCancelButton: true,
                    confirmButtonText: 'Continue',
                    cancelButtonText: 'Not now',
                    reverseButtons: true,
                });
                if (!confirmation.isConfirmed) {
                    return;
                }
            }

            try {
                const permission = Notification.permission === 'granted'
                    ? 'granted'
                    : await Notification.requestPermission();
                if (permission !== 'granted') {
                    const blocked = permission === 'denied';
                    setNearbyNotificationsEnabled(false, blocked
                        ? 'Notifications are blocked. Allow them in browser site settings.'
                        : 'Notification permission has not been granted.');
                    gpsStatus.textContent = blocked
                        ? 'Notifications are blocked for this site. Allow them in your browser site settings, then try again.'
                        : 'Notification permission was not granted.';
                    if (window.Swal) {
                        await window.Swal.fire({
                            icon: 'warning',
                            title: blocked ? 'Notifications are blocked' : 'Permission not granted',
                            text: blocked
                                ? 'Allow notifications for this site in your browser settings, then click the bell again.'
                                : 'Allow browser notifications to receive nearby brake-event and accident-prone-area alerts.',
                        });
                    }
                    return;
                }
            } catch (error) {
                console.error('Unable to request nearby notification permission.', error);
                gpsStatus.textContent = 'Unable to request browser notification permission.';
                if (window.Swal) {
                    await window.Swal.fire({
                        icon: 'error',
                        title: 'Permission request failed',
                        text: 'Your browser could not request notification permission. Check its site settings and try again.',
                    });
                }
                return;
            }

            if (!await confirmAndStartGpsTracking()) {
                gpsStatus.textContent = 'GPS tracking was not started; nearby notifications remain off.';
                return;
            }

            setNearbyNotificationsEnabled(true);
            mapNote.textContent = 'Nearby notifications are on for hard and emergency brake events and saved accident-prone areas within 400 meters. Keep the dashboard open and GPS tracking enabled.';
            try {
                const notification = new Notification('Nearby notifications enabled', {
                    body: 'You will be alerted when a hard or emergency brake event or saved accident-prone area is within 400 meters.',
                    tag: 'brakesafe-notifications-enabled',
                });
                activeNearbyNotifications.add(notification);
                notification.onclose = () => activeNearbyNotifications.delete(notification);
            } catch (error) {
                console.error('Unable to show the nearby notifications confirmation.', error);
            }
            if (window.Swal) {
                await window.Swal.fire({
                    toast: true,
                    position: 'top-end',
                    icon: 'success',
                    title: 'Nearby notifications enabled',
                    text: 'You will be alerted when a target is within 400 meters.',
                    showConfirmButton: false,
                    timer: 5000,
                    timerProgressBar: true,
                });
            }
        });

        const themeButton = document.getElementById('theme-toggle');
        themeButton.addEventListener('click', () => {
            const isLight = document.getElementById('dashboard-root').classList.toggle('dashboard-light');
            themeButton.setAttribute('aria-label', isLight ? 'Switch to dark theme' : 'Switch to light theme');
            themeButton.querySelector('span').textContent = isLight ? 'Light theme' : 'Dark theme';
        });

        if (resumeGpsAfterReload && !startGpsTracking()) {
            writeSessionFlag(gpsSessionKey, false);
        }
        if (resumeNotificationsAfterReload) {
            if (resumeGpsAfterReload && window.Notification && Notification.permission === 'granted') {
                setNearbyNotificationsEnabled(true);
            } else {
                writeSessionFlag(notificationsSessionKey, false);
            }
        }

    });
</script>

        (() => {
            const transitionKey = 'brakesafe.pageTransition';
            const startTransition = () => {
                const transition = document.getElementById('brakesafe-page-transition');
                let shouldAnimate = false;
                try {
                    shouldAnimate = sessionStorage.getItem(transitionKey) === '1';
                    sessionStorage.removeItem(transitionKey);
                } catch (error) {
                    console.error('Unable to read the page transition state.', error);
                }

                document.documentElement.classList.remove('page-transition-pending');
                document.body.style.removeProperty('visibility');
                if (!shouldAnimate || !transition) return;

                transition.hidden = false;
                requestAnimationFrame(() => transition.classList.add('is-active'));
                window.setTimeout(() => {
                    transition.classList.remove('is-active');
                    window.setTimeout(() => { transition.hidden = true; }, 350);
                }, window.matchMedia('(prefers-reduced-motion: reduce)').matches ? 650 : 3600);
            };

            document.addEventListener('click', (event) => {
                if (event.defaultPrevented || event.button !== 0
                    || event.metaKey || event.ctrlKey || event.shiftKey || event.altKey
                    || !(event.target instanceof Element)) return;

                const link = event.target.closest('a[href]');
                if (!link || link.target && link.target !== '_self' || link.hasAttribute('download')) return;

                const destination = new URL(link.href, window.location.href);
                if (destination.origin !== window.location.origin
                    || destination.pathname === window.location.pathname
                    && destination.search === window.location.search) return;

                try {
                    sessionStorage.setItem(transitionKey, '1');
                } catch (error) {
                    console.error('Unable to save the page transition state.', error);
                }
            }, true);

            startTransition();
        })();

            const distanceInMeters = (latitudeA, longitudeA, latitudeB, longitudeB) => {
                const radians = (degrees) => degrees * Math.PI / 180;
                const latitudeDelta = radians(latitudeB - latitudeA);
                const longitudeDelta = radians(longitudeB - longitudeA);
                const haversine = Math.sin(latitudeDelta / 2) ** 2
                    + Math.cos(radians(latitudeA)) * Math.cos(radians(latitudeB))
                    * Math.sin(longitudeDelta / 2) ** 2;
                return 6371000 * 2 * Math.atan2(Math.sqrt(haversine), Math.sqrt(1 - haversine));
            };
